Sơ đồ thể hiện luồng liên kết giữa 3 Service. Kéo chuột để di chuyển (Pan) và dùng nút hoặc lăn chuột để phóng to/thu nhỏ (Zoom).

    %% ──────────────────────────────────────────
    %% 1. IDENTITY SERVICE DOMAIN
    %% ──────────────────────────────────────────
    users {
        uuid id PK "users.id (Canonical UUID)"
        string keycloak_id UK "Keycloak Sub (36 chars)"
        string username UK
        string email UK
        string first_name
        string last_name
        string status "ACTIVE, INACTIVE, SUSPENDED"
        timestamp created_at
        timestamp updated_at
    }

    user_roles {
        uuid user_id FK "users.id"
        string role "ROLE_USER, ROLE_ADMIN"
    }

    verification_tokens {
        uuid id PK
        uuid user_id FK "users.id"
        string token UK
        string type "EMAIL_VERIFY, PASSWORD_RESET"
        timestamp expires_at
        timestamp used_at
    }

    users ||--o{ user_roles : "has"
    users ||--o{ verification_tokens : "owns"

    %% ──────────────────────────────────────────
    %% 2. WORKSPACE SERVICE DOMAIN (ROOT TENANT)
    %% ──────────────────────────────────────────
    workspaces {
        uuid id PK "Gốc Tenant duy nhất"
        string name "Tên không gian"
        string slug UK "Đường dẫn định danh"
        text description
        string avatar_url
        uuid created_by "FK logic -> users.id"
        timestamp created_at
        timestamp updated_at
        timestamp deleted_at "Soft Delete support"
    }

    workspace_members {
        uuid id PK
        uuid workspace_id FK "workspaces.id"
        uuid user_id "FK logic -> users.id"
        string role "OWNER, ADMIN, MEMBER, GUEST"
        uuid invited_by
        timestamp joined_at
    }

    workspace_invitations {
        uuid id PK
        uuid workspace_id FK "workspaces.id"
        uuid invitee_user_id "FK logic -> users.id"
        uuid invited_by "FK logic -> users.id"
        string role "OWNER, ADMIN, MEMBER, GUEST"
        string status "PENDING, ACCEPTED, DECLINED, CANCELLED"
        timestamp created_at
        timestamp responded_at
    }

    workspaces ||--o{ workspace_members : "contains"
    workspaces ||--o{ workspace_invitations : "invites"

    %% ──────────────────────────────────────────
    %% 3. WORK SERVICE: PROJECT & RBAC CORE

        uuid id PK
        uuid workspace_id "FK logic -> workspaces.id"

        string key UK "Mã tiền tố (Jira Key)"

        string template_category

        date end_date
        uuid created_by
        timestamp created_at
        timestamp deleted_at

    project_roles {
        uuid id PK

        string name "ADMINISTRATOR, MEMBER, VIEWER"
        text description

    role_assignments {
        uuid id PK
        uuid project_id FK "projects.id"
        uuid user_id "FK logic -> users.id"
        uuid role_id FK "project_roles.id"
        timestamp assigned_at
    }

    permission_scheme_entries {
        uuid id PK
        uuid project_id FK "projects.id"
        uuid role_id FK "project_roles.id"
        string permission_key "ISSUE_CREATE, SPRINT_MANAGE..."

    invitations {
        uuid id PK
        string type "PROJECT (Loại bỏ TEAM)"
        uuid target_id "projects.id"
        uuid invitee_user_id "users.id"
        uuid invited_by "users.id"
        uuid project_role_id FK "project_roles.id"
        string status "PENDING, ACCEPTED, DECLINED"
    }

    project_clones {
        uuid id PK
        uuid source_project_id FK "projects.id"
        uuid cloned_project_id FK "projects.id"
        string clone_type "FROM_TEMPLATE, DUPLICATE"

    projects ||--o{ project_roles : "defines"
    projects ||--o{ role_assignments : "assigns"
    project_roles ||--o{ role_assignments : "has"

    project_roles ||--o{ permission_scheme_entries : "grants"
    projects ||--o{ invitations : "receives"
    projects ||--o{ project_clones : "clones"

        uuid project_id FK "projects.id"

        string category "TODO, IN_PROGRESS, DONE"
        string color_hex
    }

    sprints {
        uuid id PK
        uuid project_id FK "projects.id"
        string name
        text goal
        string state "FUTURE, ACTIVE, CLOSED"
        timestamp start_date
        timestamp end_date
        timestamp completed_date
    }

    work_items {
        uuid id PK
        uuid project_id FK "projects.id"
        uuid type_id FK "work_item_types.id"
        uuid status_id FK "statuses.id"
        uuid parent_id FK "work_items.id"
        uuid sprint_id FK "sprints.id"
        string code "KEY-123"
        string summary
        text description
        string priority "LOW, MEDIUM, HIGH, URGENT"
        uuid reporter_id "users.id"
        uuid assignee_id "users.id"
        decimal story_points
        string rank "LexoRank ordering"
        jsonb custom_fields
        date due_date
        int version "Optimistic Locking"

    boards {
        uuid id PK
        uuid project_id FK "projects.id"
        string name "Scrum Board / Kanban Board"
        string type "SCRUM, KANBAN"
        text filter_jql
    }

    board_columns {
        uuid id PK
        uuid board_id FK "boards.id"
        string name "To Do, Doing, Review, Done"
        int column_order
        int min_wip_limit
        int max_wip_limit
    }

    board_column_statuses {
        uuid column_id PK,FK "board_columns.id"
        uuid status_id PK,FK "statuses.id"

    projects ||--o{ work_items : "owns"
    work_item_types ||--o{ work_items : "types"
    statuses ||--o{ work_items : "states"
    sprints ||--o{ work_items : "contains"
    work_items ||--o{ work_items : "parent_of"

    projects ||--o{ boards : "has"
    boards ||--o{ board_columns : "structures"
    board_columns ||--o{ board_column_statuses : "maps"
    statuses ||--o{ board_column_statuses : "mapped_in"

⚖️ So Sánh Kiến Trúc Dữ Liệu (Trước vs Sau Refactor)

❌ Mô Hình Cũ (Có Khái Niệm Team)
  • Cấu trúc 3 cấp cồng kềnh: Workspace ➔ Team ➔ Project ➔ Issue. Người dùng bắt buộc phải tạo Team trước khi được tạo bất kỳ Dự án nào.
  • Xung đột nguồn sự thật quyền hạn: Tồn tại đồng thời teams.owner_id, team_members.role (LEADER) và role_assignments gây ra ~6 helper phân xử phức tạp và dễ phát sinh lỗ hổng bảo mật.
  • Khóa ngoại gián tiếp: Bảng projects mang team_id REFERENCES teams(id). Khi cần lọc dự án theo Workspace, hệ thống phải JOIN qua bảng teams (N+1 query nguy hiểm).
  • Dư thừa bảng: Tồn tại bảng teams và team_members với chi phí bảo trì cao nhưng không đem lại giá trị thực tế cho quy trình Agile.
✅ Mô Hình Mới (Phẳng - Jira Cloud Standard)
  • Cấu trúc 2 cấp tinh gọn: Workspace ➔ Project ➔ Work Items. Người dùng tạo Dự án tức thì ngay trong Workspace mà không cần bước đệm Team.
  • Nguồn sự thật phân quyền duy nhất: Toàn bộ thành viên và quyền hạn dự án được tập trung tại role_assignments + project_roles + permission_scheme_entries.
  • Truy vấn tốc độ cao (Index-backed): Bảng projects mang trực tiếp workspace_id kèm chỉ mục idx_projects_workspace. Lấy danh sách dự án trong 1 câu SELECT, zero JOIN thừa.
  • Tự động Onboarding: Người dùng mới đăng ký luôn được tự động khởi tạo sẵn một Workspace mặc định và chuyển thẳng vào làm việc, không còn màn hình trống.
